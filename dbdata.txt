model Product {
id           String   @default(cuid()) @id
name         String
category     String
price        Float
description  String?
thumbnail    String?
createdDate  DateTime @default(now())
updatedDate  DateTime @updatedAt
}

model ProductImage {
id        String  @default(cuid()) @id
productId String  @relation(fields: [productId], references: [id])
thumbnailUrl  String
imageUrl1 String?
imageUrl2 String?
imageUrl3 String?
imageUrl4 String?
imageUrl5 String?
imageUrl6 String?
}

model NutritionalInfo {
id             String  @default(cuid()) @id
productId      String  @relation(fields: [productId], references: [id])
energy         Float?
protein        Float?
fat            Float?
carbohydrates  Float?
saltEquivalent Float?
allergen  String?
}
